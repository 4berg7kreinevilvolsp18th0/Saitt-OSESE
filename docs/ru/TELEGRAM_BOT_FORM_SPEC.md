# Спецификация формы обращений для Telegram-бота

**Дата:** 2024-12-19  
**Версия:** 1.0

---

## 📋 Обзор

Спецификация формы для подачи обращений через Telegram-бота. Форма полностью соответствует структуре формы на сайте, но адаптирована для Telegram Bot API с поддержкой анонимной подачи.

---

## 🎯 Требования

### Обязательные поля
- ✅ Тема обращения (title)
- ✅ Описание (description)
- ✅ Школа/Институт (institute)

### Опциональные поля
- ⭕ Направление (direction)
- ⭕ Контакт для связи (contact)
- ⭕ Вложения (attachments)

### Анонимная подача
- ✅ Полностью анонимная подача (без контакта)
- ✅ Контакт запрашивается только при неанонимной подаче

## 🔄 Процесс заполнения

### Схема состояний (state machine)

```
START
  ↓
SELECT_ANONYMITY (анонимно/с контактом)
  ↓
SELECT_SCHOOL (выбор школы)
  ↓
SELECT_DIRECTION (выбор направления, опционально)
  ↓
ENTER_TITLE (ввод темы)
  ↓
ENTER_DESCRIPTION (ввод описания)
  ↓
[Если не анонимно] → ENTER_CONTACT (ввод контакта)
[Если анонимно]    → ADD_ATTACHMENTS (вложения, опционально)
  ↓
ADD_ATTACHMENTS (вложения, опционально)
  ↓
PREVIEW (предпросмотр)
  ↓
CONFIRM (подтверждение)
  ↓
SUBMIT (отправка на API)
  ↓
SUCCESS (успешная отправка)
```

### Возможность возврата назад

На каждом шаге доступна кнопка "◀️ Назад" для возврата к предыдущему шагу с сохранением введенных данных.

## 🔐 Анонимная подача

### Особенности анонимной подачи

1. **Без контакта:**
   - `is_anonymous = true`
   - `contact_type = null`
   - `contact_value = null`

2. **Связь с пользователем:**
   - Используется Telegram User ID только для связи обращений одного пользователя
   - User ID НЕ сохраняется в обращении
   - Пользователь получает `public_token` для проверки статуса

3. **Ограничения:**
   - Невозможно связаться с пользователем для уточнения деталей
   - Ответ придет через бота, если пользователь проверит статус по токену

        "Выберите вашу школу или институт:\n\n"
        "Для удобства поиска используйте кнопки ниже. Вы можете выбрать из списка всех школ ДВФУ.",
        reply_markup=разметка
    )
    return ВЫБОР_НАПРАВЛЕНИЯ

async def выбор_направления(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Выбор направления обращения (опционально)"""
    запрос = update.callback_query
    данные_формы = context.user_data.get('form_data', {})
    
    # Обработка выбора школы
    if запрос.data.startswith("школа_"):
        код_школы = запрос.data.replace("школа_", "")
        данные_формы['institute'] = код_школы
        # Показываем полное название для подтверждения
        школа = next(ш for ш in ШКОЛЫ if ш["code"] == код_школы)
        await запрос.answer(f"✅ Выбрано: {школа['full']}")
    
    # Обработка кнопки "Назад"
    if запрос.data == "назад":
        return await выбор_анонимности(update, context)
    
    клавиатура = [
        [InlineKeyboardButton("⚖️ Правовой комитет", callback_data="направление_legal")],
        [InlineKeyboardButton("🏗️ Инфраструктура", callback_data="направление_infrastructure")],
        [InlineKeyboardButton("💰 Стипендиальный", callback_data="направление_scholarship")],
        [InlineKeyboardButton("🌍 Международный", callback_data="направление_international")],
        [InlineKeyboardButton("❌ Пропустить", callback_data="направление_пропустить")],
        [InlineKeyboardButton("◀️ Назад", callback_data="назад")],
    ]
    разметка = InlineKeyboardMarkup(клавиатура)
    
    await запрос.edit_message_text(
        "Выберите направление обращения (необязательно):\n\n"
        "Это поможет быстрее направить ваше обращение нужному специалисту.\n\n"
        "Если не уверены, можете пропустить этот шаг.",
        reply_markup=разметка
    )
    return ВВОД_ТЕМЫ

async def ввод_темы(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Ввод темы обращения"""
    запрос = update.callback_query
    данные_формы = context.user_data.get('form_data', {})
    
    # Обработка выбора направления
    if запрос.data.startswith("направление_"):
        направление = запрос.data.replace("направление_", "")
        if направление == "пропустить":
            данные_формы['direction'] = None
        else:
            данные_формы['direction'] = направление
        await запрос.answer()
    
    # Обработка кнопки "Назад"
    if запрос.data == "назад":
        return await выбор_школы(update, context)
    
    await запрос.edit_message_text(
        "📝 Введите тему обращения:\n\n"
        "Кратко опишите суть вашего вопроса (минимум 5 символов).\n\n"
        "Например:\n"
        "• Проблема с доступом к Wi-Fi\n"
        "• Вопрос о стипендиях\n"
        "• Недостаток мест в библиотеке"
    )
    return ВВОД_ОПИСАНИЯ

async def ввод_описания(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Ввод описания обращения"""
    текст = update.message.text
    
    # Валидация темы
    if len(текст) < 5:
        await update.message.reply_text(
            "❌ Тема должна быть не менее 5 символов.\n\n"
            "Пожалуйста, введите тему обращения еще раз."
        )
        return ВВОД_ТЕМЫ
    
    данные_формы = context.user_data.get('form_data', {})
    данные_формы['title'] = текст
    
    await update.message.reply_text(
        "📄 Опишите ситуацию подробно:\n\n"
        "Расскажите детали вашего обращения. Чем подробнее описание, тем быстрее мы сможем помочь (минимум 20 символов)."
    )
    
    # Если анонимно - пропускаем ввод контакта
    if данные_формы.get('is_anonymous'):
        return ДОБАВЛЕНИЕ_ВЛОЖЕНИЙ
    else:
        return ВВОД_КОНТАКТА

async def ввод_контакта(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Ввод контакта для связи"""
    текст = update.message.text
    
    # Валидация описания
    if len(текст) < 20:
        await update.message.reply_text(
            "❌ Описание должно быть не менее 20 символов.\n\n"
            "Пожалуйста, опишите ситуацию подробнее."
        )
        return ВВОД_ОПИСАНИЯ
    
    данные_формы = context.user_data.get('form_data', {})
    данные_формы['description'] = текст
    
    # Валидация email или telegram
    паттерн_email = r'^[\w\.-]+@[\w\.-]+\.\w+$'
    паттерн_telegram = r'^@[\w]+$'
    
    if re.match(паттерн_email, текст):
        данные_формы['contact_type'] = 'email'
        данные_формы['contact_value'] = текст
    elif re.match(паттерн_telegram, текст):
        данные_формы['contact_type'] = 'telegram'
        данные_формы['contact_value'] = текст
    else:
        await update.message.reply_text(
            "❌ Укажите email (example@mail.ru) или Telegram (@username)\n\n"
            "Пожалуйста, введите контакт в правильном формате."
        )
        return ВВОД_КОНТАКТА
    
    клавиатура = [
        [InlineKeyboardButton("✅ Пропустить", callback_data="пропустить_вложения")],
        [InlineKeyboardButton("◀️ Назад", callback_data="назад")],
    ]
    разметка = InlineKeyboardMarkup(клавиатура)
    
    await update.message.reply_text(
        "📎 Добавить вложения (опционально):\n\n"
        "Вы можете приложить до 5 файлов:\n"
        "• Изображения (JPG, PNG)\n"
        "• Документы (PDF, DOC, DOCX)\n"
        "• Максимальный размер файла: 20 MB\n\n"
        "Отправьте файлы или нажмите 'Пропустить'",
        reply_markup=разметка
    )
    return ДОБАВЛЕНИЕ_ВЛОЖЕНИЙ

обработчик_формы = ConversationHandler(
